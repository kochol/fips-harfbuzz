# Static shaping library with FreeType interop and no optional UI dependencies.
cmake_minimum_required(VERSION 3.21)
if(TARGET harfbuzz)
    return()
endif()

if(NOT TARGET freetype)
    if(COMMAND fips_import_fips_freetype_freetype)
        fips_import_fips_freetype_freetype()
    else()
        message(FATAL_ERROR "Import fips-freetype before fips-harfbuzz.")
    endif()
endif()

set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
set(BUILD_SHARED_LIBS OFF)
set(HB_HAVE_FREETYPE ON)
set(HB_HAVE_ICU OFF)
set(HB_HAVE_CAIRO OFF)
set(HB_HAVE_GLIB OFF)
set(HB_HAVE_GRAPHITE2 OFF)
set(HB_HAVE_GOBJECT OFF)
set(HB_HAVE_INTROSPECTION OFF)
set(HB_HAVE_CORETEXT OFF)
set(HB_HAVE_UNISCRIBE OFF)
set(HB_HAVE_DIRECTWRITE OFF)
set(HB_HAVE_GDI OFF)
set(HB_BUILD_UTILS OFF)
set(HB_BUILD_SUBSET OFF)
set(HB_BUILD_RASTER OFF)
set(HB_BUILD_VECTOR OFF)
set(HB_BUILD_GPU OFF)
set(HB_BUILD_GPU_DEMO OFF)
set(BUILD_TESTING OFF)
add_subdirectory("../.." "${CMAKE_CURRENT_BINARY_DIR}/upstream")
if(NOT TARGET harfbuzz::harfbuzz)
    add_library(harfbuzz::harfbuzz ALIAS harfbuzz)
endif()
